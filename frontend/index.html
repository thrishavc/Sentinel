SELECT 
    f.fir_no, f.district, f.section_code,
    c.chargesheet_no, c.status AS chargesheet_status,
    a.arrest_date, a.jail_status,
    m.mo_match_score, m.pattern_type
FROM cctns_fir_register f
LEFT JOIN cctns_chargesheet_v2 c ON f.fir_id = c.fir_id
LEFT JOIN iapps_arrest_log a ON f.accused_id = a.accused_id
LEFT JOIN mo_matching_index m ON f.mo_code = m.mo_code
WHERE f.accused_id = 'ACC-89241' AND f.district IN ('Whitefield', 'Ramanagara');
